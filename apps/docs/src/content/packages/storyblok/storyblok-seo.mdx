# SEO

Handling the SEO.

`getStoryblokSeoData` contains main functionality of the package. It gets SEO data from Storyblok and generates
metadata object which can be used in the Next page. See
[docs about using metadata in Next.js ↗](https://nextjs.org/docs/app/api-reference/functions/generate-metadata)
for more datails.

<Callout type="info" emoji="ℹ️">
  Used in the [web app](/apps/web).
</Callout>

## Usage

<Steps>
  ### Import package in package.json

```json
"dependencies": {
    "@natu/storyblok": "workspace:*",
}
```

### Generate metadata object where needed

e.g. `layout.tsx` file, `page.tsx` file etc.

```tsx filename="apps/web/src/app/layout.tsx" /getStoryblokSeoData/
import { Metadata } from 'next';
import { draftMode } from 'next/headers';
import { getStoryblokSdk } from '@natu/storyblok/api';
import { getStoryblokSeoData } from '@natu/storyblok/getStoryblokSeoData';

export const generateMetadata = async (): Promise<Metadata> => {
  const { isEnabled } = await draftMode();
  const { getConfigNode } = getStoryblokSdk({ draftMode: isEnabled });

  const { data } = await getConfigNode();

  const configContent = data?.story?.content;

  return getStoryblokSeoData(configContent?.defaultSeo, {
    slug: '/',
    siteName: configContent?.siteName || '',
  });
};
```

```tsx filename="apps/web/src/app/[...slug]/page.tsx" /getStoryblokSeoData/
import { Metadata } from 'next';
import { draftMode } from 'next/headers';
import { getStoryblokSdk } from '@natu/storyblok/api';
import { getStoryblokSeoData } from '@natu/storyblok/getStoryblokSeoData';

export const generateMetadata = async (
  { params }: PageProps,
  parent: ResolvingMetadata,
): Promise<Metadata> => {
  const { isEnabled } = await draftMode();
  const { getContentNode } = getStoryblokSdk({ draftMode: isEnabled });

  const awaitedParas = await params;

  const slug = getSlugWithAppName({ slug: getSlugFromParams(awaitedParas.slug) });

  const prevData = await parent;
  const { data } = await tryCatch(
    getContentNode({
      slug,
    }),
  );

  return getStoryblokSeoData(data?.data?.story?.content?.seo, {
    slug: `/${getSlugFromParams(awaitedParas.slug)}`,
    prevData,
  });
};
```

</Steps>

## Sitemap.xml

The application also builds a sitemap of pages, which is accessible at this address `/sitemap.xml`

Check the const `validSitemapComponents` in `sitemap.ts` file. These are the content types that are to be included in the sitemap. For example, if you add a blog, just add the name of the new content type after the comma:

```ts
// add more -> "page,other-content-type,other"
const validSitemapComponents = 'page,blogPost';
```

---

If a page has the `noIndex` select option checked, it will not be included in the sitemap.
